local pages, tabButtons = {}, {}
local function MakeTab(name)
	local btn = Create("TextButton",{Size=UDim2.new(1,0,0,30),BackgroundColor3=Theme.panel2,Text="  "..name,
		TextColor3=Theme.sub,Font=Enum.Font.GothamMedium,TextSize=13,TextXAlignment=Enum.TextXAlignment.Left,
		AutoButtonColor=false},Sidebar)
	Corner(btn,6)
	local page = Create("ScrollingFrame",{Size=UDim2.new(1,0,1,0),BackgroundTransparency=1,CanvasSize=UDim2.new(0,0,0,0),
		ScrollBarThickness=4,Visible=false},Content)
	Create("UIListLayout",{Padding=UDim.new(0,6),SortOrder=Enum.SortOrder.LayoutOrder},page)
	Pad(page,2)
	pages[name]=page; tabButtons[name]=btn
	btn.MouseButton1Click:Connect(function()
		for n,p in pairs(pages) do p.Visible=(n==name) end
		for n,b in pairs(tabButtons) do
			b.BackgroundColor3=(n==name) and Theme.accent or Theme.panel2
			b.TextColor3=(n==name) and Theme.text or Theme.sub
		end
	end)
	page.UIListLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		page.CanvasSize = UDim2.new(0,0,0,page.UIListLayout.AbsoluteContentSize.Y+10)
	end)
	return page
end

local function Selector(p,t,list,cb,btnText,btnCb)
	local row=Create("Frame",{Size=UDim2.new(1,0,0,30),BackgroundColor3=Theme.panel,BorderSizePixel=0},p); Corner(row,6)
	Create("TextLabel",{Size=UDim2.new(0,70,0,30),Position=UDim2.new(0,8,0,0),BackgroundTransparency=1,Text=t,
		TextColor3=Theme.sub,Font=Enum.Font.Gotham,TextSize=12,TextXAlignment=Enum.TextXAlignment.Left},row)
	local idx=1
	local item=Create("TextLabel",{Size=UDim2.new(0,170,0,30),Position=UDim2.new(0,74,0,0),BackgroundTransparency=1,
		Text=(list[1] or "—"),TextColor3=Theme.text,Font=Enum.Font.GothamMedium,TextSize=12,
		TextXAlignment=Enum.TextXAlignment.Left,TextTruncate=Enum.TextTruncate.AtEnd},row)
	local prev=Create("TextButton",{Size=UDim2.new(0,22,0,22),Position=UDim2.new(1,-134,0,4),BackgroundColor3=Theme.panel2,
		Text="<",TextColor3=Theme.text,Font=Enum.Font.GothamBold},row)
	local nxt=Create("TextButton",{Size=UDim2.new(0,22,0,22),Position=UDim2.new(1,-110,0,4),BackgroundColor3=Theme.panel2,
		Text=">",TextColor3=Theme.text,Font=Enum.Font.GothamBold},row)
	Corner(prev,5); Corner(nxt,5)
	local act
	if btnText then act=Create("TextButton",{Size=UDim2.new(0,80,0,22),Position=UDim2.new(1,-84,0,4),
		BackgroundColor3=Theme.accent,Text=btnText,TextColor3=Theme.text,Font=Enum.Font.GothamBold,TextSize=11},row); Corner(act,5)
	act.MouseButton1Click:Connect(function() if list[idx] and btnCb then pcall(function() btnCb(list[idx]) end) end end) end
	local function render() item.Text=list[idx] or "—"; if list[idx] then cb(list[idx]) end end
	prev.MouseButton1Click:Connect(function() idx=idx-1; if idx<1 then idx=#list end; render() end)
	nxt.MouseButton1Click:Connect(function() idx=idx+1; if idx>#list then idx=1 end; render() end)
	render()
	return {SetList=function(l) list=l; idx=1; render() end, Get=function() return list[idx] end}
end

--============================== HELPERS ======================================
local TweenObj
local function HRP() local c=LP.Character; return c and c:FindFirstChild("HumanoidRootPart") end
local function Char() return LP.Character end
local function TP(cf)
	pcall(function()
		local hrp=HRP(); if not hrp then return end
		if Config.InstantTP then
			if TweenObj then TweenObj:Cancel() end
			hrp.CFrame=cf
		else
			if TweenObj then TweenObj:Cancel() end
			local d=(hrp.Position-cf.Position).Magnitude
			TweenObj=TweenService:Create(hrp,TweenInfo.new(math.max(d/500,0.01),Enum.EasingStyle.Linear),{CFrame=cf})
			TweenObj:Play()
		end
	end)
end
pcall(function() sethiddenproperty(LP,"SimulationRadius",math.huge) end)
local function Notify(t,x) pcall(function() StarterGui:SetCore("SendNotification",{Title=t,Text=x,Duration=5}) end) end
local function Lvl() local d=LP:FindFirstChild("Data"); return (d and d:FindFirstChild("Level") and d.Level.Value) or 1 end

local function EquipWeapon()
	local c=Char(); if not c then return end
	local name = Config.Weapon
	local bp=LP:FindFirstChild("Backpack")
	if name and bp and bp:FindFirstChild(name) and c:FindFirstChild("Humanoid") then
		pcall(function() c.Humanoid:EquipTool(bp[name]) end); return
	end
	if Config.AutoEquip and not c:FindFirstChildOfClass("Tool") and bp then
		local t=bp:FindFirstChildOfClass("Tool")
		if t and c:FindFirstChild("Humanoid") then pcall(function() c.Humanoid:EquipTool(t) end) end
	end
end

local function GetQuest(level)
	local set
	if game.PlaceId==2753915549 then set=Sea1
	elseif game.PlaceId==4442272183 then set=Sea2
	elseif game.PlaceId==7449423635 then set=Sea3 end
	if not set then return nil end
	for _,q in ipairs(set) do if level>=q[1] and level<=q[2] then return q end end
	return set[#set]
end

local function FindEntity(filter, useBossList)
	local hrp=HRP(); if not hrp then return end
	local containers={Workspace}
	local en=Workspace:FindFirstChild("Enemies"); if en then table.insert(containers,en) end
	local best,dist
	for _,cont in ipairs(containers) do
		for _,v in pairs(cont:GetChildren()) do
			if v:FindFirstChild("Humanoid") and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health>0 then
				local match=false
				if useBossList then
					for _,b in ipairs(BossList[game.PlaceId] or {}) do
						if string.find(string.lower(v.Name), string.lower(b), 1, true) then match=true break end
					end
				elseif filter then
					match = string.find(string.lower(v.Name), string.lower(filter), 1, true) ~= nil
				end
				if match then
					local d=(v.HumanoidRootPart.Position-hrp.Position).Magnitude
					if not dist or d<dist then best,dist=v,d end
				end
			end
		end
	end
	return best
end

local function AttackOnce()
	pcall(function()
		local ac=CF and CF.activeController
		if ac then
			ac.attacking=false; ac.blocking=false
			ac.timeToNextAttack=tick()-1; ac.timeToNextBlock=0
			ac.increment=3; ac.hitboxMagnitude=Config.Hitbox
		end
		local c=Char()
		if c then
			if c:FindFirstChild("Stun") then c.Stun.Value=0 end
			if c:FindFirstChild("Humanoid") then c.Humanoid.Sit=false end
		end
		VirtualUser:CaptureController()
		VirtualUser:Button1Down(Vector2.new(1280,672))
	end)
end

end

local function MagnetTo(target)
	local hrp=HRP(); if not hrp or not target then return end
	local o=target.HumanoidRootPart
	if (o.Position-hrp.Position).Magnitude < Config.MagnetRadius then
		o.CFrame = hrp.CFrame * CFrame.new(0,25,0)
		o.Size=Vector3.new(30,30,30); o.Transparency=0.7; o.CanCollide=false
		if target:FindFirstChild("Humanoid") then target.Humanoid.PlatformStand=true end
		if Config.AnchorMobs then o.Anchored=true end
	end
end

RunService.Heartbeat:Connect(function()
	if not (Config.Magnet and (Config.AutoFarm or Config.AutoBoss or Config.AutoBlackbeard or Config.AutoMastery)) then return end
	local hrp=HRP(); if not hrp then return end
	if Config.AutoFarm or Config.AutoMastery then
		local q=GetQuest(Lvl()); if not q then return end
		local en=Workspace:FindFirstChild("Enemies"); if not en then return end
		local n=0
		for _,v in pairs(en:GetChildren()) do
			if v.Name==q[3] and v:FindFirstChild("Humanoid") and v.Humanoid.Health>0 then
				MagnetTo(v); n=n+1; if n>=Config.MaxMobs then break end
			end
		end
	end
	if Config.AutoBoss then local b=FindEntity(nil,true); if b then MagnetTo(b) end end
	if Config.AutoBlackbeard then local b=FindEntity("Blackbeard"); if b then MagnetTo(b) end end
end)

								end

						end
						if not found then TP(q[8]) end

			if c and not c:FindFirstChild("HasBuso") then pcall(function() CommF:InvokeServer("Buso") end) end

	end
end)
